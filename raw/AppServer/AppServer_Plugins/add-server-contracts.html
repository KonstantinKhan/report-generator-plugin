Получение доступа к API сервера

Добавьте публичный конструктор класса.

В качестве аргументов конструктор может принимать интерфейсы API сервера, доступные через механизм внедрения зависимостей.

Подробнее о механизме внедрения зависимостей - https://learn.microsoft.com/ru-ru/dotnet/core/extensions/dependency-injection

В нашем примере это IBProcAttributeMetaDataContract из пространства имен Ascon.Plm.AppServer.Contracts.ConfigServer - интерфейс предоставляющий доступ к методам взаимодействия с атрибутами бизнес-процесса. 

using Asp.Versioning;
using Microsoft.AspNetCore.Mvc;

using Ascon.Plm.AppServer.WebApi.Controllers;
using Ascon.Plm.AppServer.Contracts.ConfigServer;

namespace AppServer.Plugin.Sample
{
    [ApiController]
    [Route("api/v{version:ApiVersion}/[controller]")]
    [ApiVersion("3.0")]
    public class SampleContoller : ApiController
    {
        private readonly IBProcAttributeMetaDataContract _attributeMetaDataContract;
        public SampleContoller(IBProcAttributeMetaDataContract attributeMetaDataContract)
        {
            _attributeMetaDataContract = attributeMetaDataContract;
        }
    }
}
